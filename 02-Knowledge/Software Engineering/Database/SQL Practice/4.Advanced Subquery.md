Q58. Cho biết tên giáo viên nào mà tham gia đề tài đủ tất cả các chủ đề.

- Dùng COUNT
```sql
SELECT GV.HOTEN
FROM GIAOVIEN GV
	JOIN THAMGIADETAI TGDT ON GV.MAGV = TGDT.MAGV
    JOIN DETAI DT ON TGDT.MADT = DT.MADT
    JOIN CHUDE CD ON DT.MACD = CD.MACD
GROUP BY GV.MAGV, GV.HOTEN
HAVING COUNT(DISTINCT CD.MACD) = (
	SELECT COUNT(*)
    FROM CHUDE
)
```

- Dùng NOT EXIST, Đi tìm giáo viên mà không tồn tại một cái chủ đề nào mà giáo viên đó chưa tham gia. Tức là nếu giáo viên đó đã tham gia tất cả chủ đề rồi thì mình select cái đống chưa tham gia phải là empty.
```sql
SELECT GV.HOTEN
FROM GIAOVIEN GV
WHERE NOT EXISTS (
	# Kiểm tra xem có cái chủ đề nào mà giáo viên này chưa làm hay không
	# Nếu không có cái nào thì có nghĩa là đã tham gia hết rồi
	SELECT 1
    FROM CHUDE CD
    WHERE CD.MACD NOT IN (
	    # Tập chủ đề GV đã tham gia
		SELECT DISTINCT CD1.MACD
        FROM GIAOVIEN GV1
			JOIN THAMGIADETAI TGDT ON GV1.MAGV = TGDT.MAGV
			JOIN DETAI DT ON TGDT.MADT = DT.MADT
			JOIN CHUDE CD1 ON DT.MACD = CD1.MACD
		WHERE GV.MAGV = GV1.MAGV
    )
)

```

- Dùng NOT EXISTS
```sql
SELECT GV.HOTEN
FROM GIAOVIEN GV
WHERE NOT EXISTS (
	# Tìm xem liệu có cái chủ đề nào mà giáo viên này chưa tham gia hay chưa
    SELECT 1 FROM CHUDE CD
    WHERE NOT EXISTS (
        SELECT 1
        FROM THAMGIADETAI TGDT
            JOIN DETAI DT ON TGDT.MADT = DT.MADT
        WHERE TGDT.MAGV = GV.MAGV
          AND DT.MACD = CD.MACD
    )
)
```
